        function renderPlanContent() {
            const container = document.getElementById('planContent');
            const day = planData[currentDay];

            const total = day.tasks.length;
            const completed = day.tasks.filter(t => typeof t === 'object' && t.completed).length;
            const percent = total > 0 ? Math.round((completed / total) * 100) : 0;

                <div class="card-header">

                    <p>${day.desc}</p>

                        <div class="progress-fill" style="width: ${percent}%;"></div>
                    </div>
                </div>
                <div class="task-list">
            `;

                const text = typeof task === 'object' ? task.text : task;
                const isChecked = typeof task === 'object' ? task.completed : false;

                html += `
                    <div class="task-item">
                        <div class="task-left">
                            <input type="checkbox" class="task-checkbox" ${isChecked ? 'checked' : ''} onchange="togglePlanTask(${idx})">
                            <span class="task-text ${isChecked ? 'completed' : ''}">${text}</span>
                        </div>
                        <div class="task-actions">
                            <button class="action-btn edit" onclick="editPlanTask(${idx})"><i class="fa-solid fa-pen"></i></button>
                            <button class="action-btn delete" onclick="deletePlanTask(${idx})"><i class="fa-solid fa-trash"></i></button>
                        </div>
                    </div>

                    <button class="add-task-btn" onclick="addPlanTask()">+ Criar</button>
                </div>
            `;

            container.innerHTML = html;
        }

        function togglePlanTask(idx) {
            const task = planData[currentDay].tasks[idx];
            if (typeof task === 'string') {
                planData[currentDay].tasks[idx] = { text: task, completed: true };
            } else {
                task.completed = !task.completed;
            }
            saveData();
            renderPlanContent();
        }

        function editPlanTask(idx) {
            const task = planData[currentDay].tasks[idx];
            const currentText = typeof task === 'object' ? task.text : task;
            const newText = prompt("Editar tarefa:", currentText);
            if (newText !== null && newText.trim() !== "") {
                if (typeof task === 'object') task.text = newText.trim();
                else planData[currentDay].tasks[idx] = { text: newText.trim(), completed: false };
                saveData();
                renderPlanContent();
            }
        }

        function deletePlanTask(idx) {
            planData[currentDay].tasks.splice(idx, 1);
            saveData();
            renderPlanContent();
        }
